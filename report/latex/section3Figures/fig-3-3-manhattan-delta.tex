\newcommand{\board}[3]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \row [count=\r from 0] in {#3} {
      \foreach \v [count=\c from 0] in \row {
        \ifnum\v=0\relax
          \node[blank] at (\c*0.98,-\r*0.98) {};%
        \else
          \ifnum\v=8\relax
            \node[moved] at (\c*0.98,-\r*0.98) {\v};%
          \else
            \node[cell] at (\c*0.98,-\r*0.98) {\v};%
          \fi
        \fi
      }
    }
  \end{scope}}
